     4→// Source of truth: docs/planning/ERD_V1.1_WEB.md
     5→// Governing law:   docs/planning/CONSTITUTION_V1.1_WEB.md
     6→//
     7→// Global conventions (Constitution v1.1 §5, ERD v1.1 §2):
     8→//   * Primary keys:        `String @id @default(cuid())` (cuid is time-ordered)
     9→//   * Audit columns:       `createdAt DateTime @default(now())` + `updatedAt DateTime @updatedAt`
    10→//   * Soft delete:         `deletedAt DateTime?` on all updatable aggregates
    11→//   * Optimistic currency: `version Int @default(1)` on BoQDocument, BoQSection,
    12→//                          BoQItem, Project, RateAnalysis, CalculationRecord,
    13→//                          ItemLibrary, CompanyProfile, UserSettings, Subscription
    14→//   * Money & quantities:  `String` storing decimal strings — application layer
    15→//                          uses decimal.js; SQLite has no native decimal type
    16→//   * Bilingual content:   `<field>En String` + `<field>Ar String?` (never concatenated)
    17→//   * Audit logging:       every mutation writes a row to AuditLog (append-only)
    18→//
    19→// Phase 1 (BoQ module) tables are complete and enforced.
    20→// Phase 2-4 tables are declared in §8 below as RESERVED additive (declared,
    21→// not enforced in Phase 1) so the schema never needs to be rewritten.
    22→// =============================================================================

   350→  createdAt     DateTime @default(now())

   352→
   353→  user             User              @relation(fields: [uploaderUserId], references: [id], onDelete: Cascade)

   359→  @@index([sha256])

   362→// AuditLog: append-only, no updates, no deletes (enforced by application layer).
   363→// Every mutation in the system writes a row here per BR-WEB-6.
   364→model AuditLog {
   365→  id          String   @id @default(cuid())
   366→  actorUserId String?  // null for system actions
   367→  action      String   // "create" | "update" | "delete" | "restore" | "import" | "export"
   368→  entityType  String   // "Project" | "BoQDocument" | "BoQItem" | ...
   369→  entityId    String
   370→  beforeJson  String?  // JSON snapshot of row before mutation
   371→  afterJson   String?  // JSON snapshot of row after mutation
   372→  ipAddress   String?
   373→  userAgent   String?
   374→  createdAt   DateTime @default(now())
   375→
   376→  actor User? @relation(fields: [actorUserId], references: [id], onDelete: SetNull)
   377→
   378→  @@index([actorUserId, createdAt])
   379→  @@index([entityType, entityId])
   380→  @@index([createdAt])
