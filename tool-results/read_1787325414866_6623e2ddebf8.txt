     4→ * SubmittalsView (S18) — Submittals Log.
     5→ *
     6→ * Per SPEC_PHASE3_WEB.md §6 (S18): table + status timeline + status-change
     7→ * actions.
     8→ *
     9→ * Data:
    10→ *   - GET  /api/projects/[projectId]/submittals   → { submittals: Submittal[] }
    11→ *   - POST /api/projects/[projectId]/submittals   → Submittal (auto-gen ref)
    12→ *   - GET  /api/submittals/[id]                   → { submittal, events }
    13→ *   - POST /api/submittals/[id]/status            → { submittal, event }
    14→ *
    15→ * Per BR-DC2: submittal ref is auto-generated as "SUB-###" (numbering domain).
    16→ * Per BR-DC5: status workflow draft → submitted → under-review →
    17→ *             approved | approved-with-comments | rejected | revise-resubmit.
    18→ *             Final = approved | approved-with-comments | rejected.
    19→ * Per BR-DC6: due date = submittedDate + reviewPeriodDays (calendar days).
    20→ *             Overdue ⇔ asOf > due AND status is non-final and not DRAFT.
    21→ *             Overdue badge: red if overdue, amber if due within 3 days.
    22→ * Per BR-DC8: every status change writes an append-only SubmittalEvent.
    23→ *
    24→ * Empty states:
    25→ *   - No project selected → "Go to Projects" CTA.
    26→ *   - No submittals yet → "Add a submittal to start tracking reviews."
    27→ *
    28→ * Layer purity: Client Component — imports only @tanstack/react-query,
    29→ * next-intl, lucide-react, @/lib/queries, @/lib/mutations, @/lib/stores,
    30→ * shadcn/ui. No server-only code.
    31→ */

   180→// ─── Component ────────────────────────────────────────────────────────────
   181→
   182→export function SubmittalsView() {
   183→  const t = useTranslations("submittals");
   184→  const tc = useTranslations("common");
   185→  const currentProjectId = useUIStore((s) => s.currentProjectId);
   186→  const setCurrentView = useUIStore((s) => s.setCurrentView);
   187→  const qc = useQueryClient();
   188→
   189→  const [selectedId, setSelectedId] = useState<string | null>(null);
   190→  const [addOpen, setAddOpen] = useState(false);
   191→  const [newSubmittal, setNewSubmittal] = useState({
   192→    subjectEn: "",
   193→    subjectAr: "",
   194→    type: "MATERIAL" as SubmittalType,
   195→    discipline: "ARC" as Discipline,
   196→    submittedDate: todayIso(),
   197→    reviewPeriodDays: 14,
   198→  });
   199→
   200→  // Injected clock (BR-DC1) — the UI supplies "today" from the browser.
   201→  const today = todayIso();
   202→
   203→  const { data, isLoading } = useQuery<SubmittalsListResponse>({
   204→    queryKey: currentProjectId
   205→      ? queryKeys.submittals.list(currentProjectId)
   206→      : ["submittals", "list", "_disabled"],
   207→    queryFn: () =>
   208→      fetchJson<SubmittalsListResponse>(
   209→        `/api/projects/${currentProjectId}/submittals`,
   210→      ),
   211→    enabled: !!currentProjectId,
   212→  });
   213→
   214→  const submittals = data?.submittals ?? [];
   215→
   216→  // Sort submittals by ref ascending for a stable table. Declared BEFORE the
   217→  // early return below to satisfy the React rules-of-hooks lint rule.
   218→  const sortedSubmittals = useMemo(
   219→    () =>
   220→      [...submittals].sort((a, b) =>
   221→        a.ref < b.ref ? -1 : a.ref > b.ref ? 1 : 0,
   222→      ),
   223→    [submittals],
   224→  );
   225→
   226→  // Detail (events timeline) for the Sheet drawer.
   227→  const { data: detailData, isLoading: detailLoading } =
   228→    useQuery<SubmittalDetailResponse>({
   229→      queryKey: selectedId
   230→        ? queryKeys.submittals.detail(selectedId)
   231→        : ["submittals", "detail", "_disabled"],
   232→      queryFn: () =>
   233→        fetchJson<SubmittalDetailResponse>(`/api/submittals/${selectedId}`),
   234→      enabled: !!selectedId,
   235→    });
   236→
   237→  // Create submittal — ref is auto-generated by the server.
   238→  const createMutation = useMutation({
   239→    mutationFn: (vars: {
   240→      subjectEn: string;
   241→      subjectAr: string | null;
   242→      type: SubmittalType;
   243→      discipline: Discipline;
   244→      submittedDate: string | null;
   245→      reviewPeriodDays: number;
   246→    }) =>
   247→      apiPost<SubmittalApi>(
   248→        `/api/projects/${currentProjectId}/submittals`,
   249→        {
   250→          projectId: currentProjectId,
   251→          subjectEn: vars.subjectEn,
   252→          subjectAr: vars.subjectAr,
   253→          type: vars.type,
   254→          discipline: vars.discipline,
   255→          submittedDate: vars.submittedDate,
   256→          reviewPeriodDays: vars.reviewPeriodDays,
   257→          resubmissionNo: 0,
   258→          status: "DRAFT",
   259→        },
   260→      ),
   261→    onSuccess: () => {
   262→      if (!currentProjectId) return;
   263→      qc.invalidateQueries({
   264→        queryKey: queryKeys.submittals.list(currentProjectId),
   265→      });
   266→      qc.invalidateQueries({
   267→        queryKey: queryKeys.documentsDashboard.detail(currentProjectId),
   268→      });
   269→      toast.success("Submittal created");
   270→      setAddOpen(false);
   271→      setNewSubmittal({
   272→        subjectEn: "",
   273→        subjectAr: "",
   274→        type: "MATERIAL",
   275→        discipline: "ARC",
   276→        submittedDate: todayIso(),
   277→        reviewPeriodDays: 14,
   278→      });
   279→    },
   280→    onError: (err: Error) => toast.error(err.message),
   281→  });
   282→
   283→  // Change status via POST /api/submittals/[id]/status.
   284→  const statusChangeMutation = useMutation({
   285→    mutationFn: (vars: {
   286→      id: string;
   287→      toStatus: SubmittalStatus;
   288→      eventDate: string;
   289→      expectedVersion: number;
   290→      note?: string;
   291→    }) =>
   292→      apiPost<{ submittal: SubmittalApi; event: SubmittalEventApi }>(
   293→        `/api/submittals/${vars.id}/status`,
   294→        {
   295→          toStatus: vars.toStatus,
   296→          eventDate: vars.eventDate,
   297→          expectedVersion: vars.expectedVersion,
   298→          note: vars.note ?? null,
   299→        },
   300→      ),
   301→    onSuccess: (_data, vars) => {
   302→      qc.invalidateQueries({
   303→        queryKey: queryKeys.submittals.detail(vars.id),
   304→      });
   305→      if (currentProjectId) {
   306→        qc.invalidateQueries({
   307→          queryKey: queryKeys.submittals.list(currentProjectId),
   308→        });
   309→        qc.invalidateQueries({
   310→          queryKey: queryKeys.documentsDashboard.detail(currentProjectId),
   311→        });
   312→      }
   313→      toast.success("Status updated");
   314→    },
   315→    onError: (err: Error) => toast.error(err.message),
   316→  });
   317→
   318→  // ─── Empty state: no project selected ──────────────────────────────────
   319→  if (!currentProjectId) {
   320→    return (
   321→      <div className="max-w-6xl mx-auto p-8">
   322→        <h1 className="text-2xl font-semibold tracking-tight">{t("title")}</h1>
   323→        <div className="mt-8 rounded-lg border border-border border-dashed p-12 text-center">
   324→          <div className="w-10 h-10 rounded-lg bg-muted flex items-center justify-center mx-auto mb-3">
   325→            <Send className="w-5 h-5 text-muted-foreground" />
   326→          </div>
   327→          <p className="text-sm text-muted-foreground mb-4">{t("noProject")}</p>
   328→          <Button
   329→            type="button"
   330→            onClick={() => setCurrentView("projects")}
   331→            className="bg-primary text-primary-foreground hover:bg-[var(--linear-primary-hover)]"
   332→          >
   333→            {t("goToProjects")}
   334→          </Button>
   335→        </div>
   336→      </div>
   337→    );
   338→  }
   339→
   340→  function submitNewSubmittal() {
   341→    if (!newSubmittal.subjectEn.trim()) {
   342→      toast.error("Subject (EN) is required");
   343→      return;
   344→    }
   345→    createMutation.mutate({
   346→      subjectEn: newSubmittal.subjectEn.trim(),
   347→      subjectAr: newSubmittal.subjectAr.trim() || null,
   348→      type: newSubmittal.type,
   349→      discipline: newSubmittal.discipline,
   350→      submittedDate: newSubmittal.submittedDate || null,
   351→      reviewPeriodDays: newSubmittal.reviewPeriodDays,
   352→    });
   353→  }
   354→
   355→  function changeStatus(s: SubmittalApi, toStatus: SubmittalStatus) {
   356→    if (!detailData) return;
   357→    statusChangeMutation.mutate({
   358→      id: s.id,
   359→      toStatus,
   360→      eventDate: today,
   361→      expectedVersion: detailData.submittal.version,
   362→    });
   363→  }
   364→
   365→  return (
   366→    <div className="max-w-6xl mx-auto p-8 space-y-6">
   367→      {/* Header */}
   368→      <div className="flex items-end justify-between gap-4">
   369→        <div>
   370→          <h1 className="text-2xl font-semibold tracking-tight">{t("title")}</h1>
   371→          <p className="text-sm text-muted-foreground mt-1">
   372→            {submittals.length}{" "}
   373→            {submittals.length === 1 ? "submittal" : "submittals"}
   374→          </p>
   375→        </div>
   376→        <Button
   377→          type="button"
   378→          onClick={() => setAddOpen(true)}
   379→          className="bg-primary text-primary-foreground hover:bg-[var(--linear-primary-hover)] gap-1.5"
   380→        >
   381→          <Plus className="w-3.5 h-3.5" />
   382→          {t("addSubmittal")}
   383→        </Button>
   384→      </div>
   385→
   386→      {/* Body */}
   387→      {isLoading ? (
   388→        <div className="rounded-lg border border-border p-8 text-center text-sm text-muted-foreground">
   389→          {tc("loading")}
   390→        </div>
   391→      ) : sortedSubmittals.length === 0 ? (
   392→        <div className="rounded-lg border border-border border-dashed p-12 text-center">
   393→          <div className="w-10 h-10 rounded-lg bg-muted flex items-center justify-center mx-auto mb-3">
   394→            <Send className="w-5 h-5 text-muted-foreground" />
   395→          </div>
   396→          <h3 className="text-sm font-medium">{t("title")}</h3>
   397→          <p className="text-xs text-muted-foreground mt-1 max-w-sm mx-auto">
   398→            {t("empty")}
   399→          </p>
   400→          <Button
   401→            type="button"
   402→            onClick={() => setAddOpen(true)}
   403→            className="mt-4 bg-primary text-primary-foreground hover:bg-[var(--linear-primary-hover)] gap-1.5"
   404→          >
   405→            <Plus className="w-3.5 h-3.5" />
   406→            {t("addSubmittal")}
   407→          </Button>
   408→        </div>
   409→      ) : (
   410→        <div className="rounded-lg border border-border overflow-hidden">
   411→          <table className="w-full text-sm">
   412→            <thead>
   413→              <tr className="border-b border-border bg-muted/30 text-left">
   414→                <th className="px-4 py-2 font-medium text-muted-foreground w-24">
   415→                  {t("ref")}
   416→                </th>
   417→                <th className="px-4 py-2 font-medium text-muted-foreground">
   418→                  {t("subject")}
   419→                </th>
   420→                <th className="px-4 py-2 font-medium text-muted-foreground w-24">
   421→                  {t("type")}
   422→                </th>
   423→                <th className="px-4 py-2 font-medium text-muted-foreground w-20">
   424→                  {t("discipline")}
   425→                </th>
   426→                <th className="px-4 py-2 font-medium text-muted-foreground w-28">
   427→                  {t("submitted")}
   428→                </th>
   429→                <th className="px-4 py-2 font-medium text-muted-foreground w-28">
   430→                  {t("due")}
   431→                </th>
   432→                <th className="px-4 py-2 font-medium text-muted-foreground w-32">
   433→                  {t("status")}
   434→                </th>
   435→                <th className="px-4 py-2 font-medium text-muted-foreground w-24">
   436→                  {t("overdue")}
   437→                </th>
   438→                <th className="px-4 py-2 font-medium text-muted-foreground w-12 text-right">
   439→                  {tc("actions")}
   440→                </th>
   441→              </tr>
   442→            </thead>
   443→            <tbody>
   444→              {sortedSubmittals.map((s) => {
   445→                const overdueState = computeOverdueState(
   446→                  today,
   447→                  s.submittedDate,
   448→                  s.reviewPeriodDays,
   449→                  s.status,
   450→                );
   451→                return (
   452→                  <tr
   453→                    key={s.id}
   454→                    onClick={() => setSelectedId(s.id)}
   455→                    className="border-b border-border last:border-0 hover:bg-muted/30 transition-colors cursor-pointer"
   456→                  >
   457→                    <td className="px-4 py-3 font-mono text-xs text-muted-foreground">
   458→                      {s.ref}
   459→                    </td>
   460→                    <td className="px-4 py-3">
   461→                      <div className="font-medium truncate">{s.subjectEn}</div>
   462→                      {s.subjectAr && (
   463→                        <div
   464→                          className="text-xs text-muted-foreground truncate"
   465→                          dir="rtl"
   466→                        >
   467→                          {s.subjectAr}
   468→                        </div>
   469→                      )}
   470→                    </td>
   471→                    <td className="px-4 py-3 font-mono text-xs text-muted-foreground">
   472→                      {s.type}
   473→                    </td>
   474→                    <td className="px-4 py-3 font-mono text-xs text-muted-foreground">
   475→                      {s.discipline}
   476→                    </td>
   477→                    <td className="px-4 py-3 font-mono text-xs text-muted-foreground">
   478→                      {s.submittedDate ?? "—"}
   479→                    </td>
   480→                    <td className="px-4 py-3 font-mono text-xs text-muted-foreground">
   481→                      {s.submittedDate
   482→                        ? addDays(s.submittedDate, s.reviewPeriodDays)
   483→                        : "—"}
   484→                    </td>
   485→                    <td className="px-4 py-3">
   486→                      <SubmittalStatusBadge status={s.status} />
   487→                    </td>
   488→                    <td className="px-4 py-3">
   489→                      <OverdueBadge state={overdueState} />
   490→                    </td>
   491→                    <td
   492→                      className="px-4 py-3 text-right"
   493→                      onClick={(e) => e.stopPropagation()}
   494→                    >
   495→                      {/* Stop row click so the dropdown doesn't double-fire. */}
   496→                      <StatusChangeMenu
   497→                        submittal={s}
   498→                        onChange={(to) => changeStatus(s, to)}
   499→                        disabled={statusChangeMutation.isPending}
   500→                      />
   501→                    </td>
   502→                  </tr>
   503→                );
   504→              })}
   505→            </tbody>
   506→          </table>
   507→        </div>
   508→      )}
   509→
   510→      {/* Create submittal dialog */}
   511→      <Dialog open={addOpen} onOpenChange={setAddOpen}>
   512→        <DialogContent className="sm:max-w-[480px]">
   513→          <DialogHeader>
   514→            <DialogTitle>{t("addSubmittal")}</DialogTitle>
   515→            <DialogDescription>{t("empty")}</DialogDescription>
   516→          </DialogHeader>
   517→          <div className="space-y-4">
   518→            <div className="space-y-1.5">
   519→              <Label htmlFor="sub-subject-en">{t("subjectEn")}</Label>
   520→              <Input
   521→                id="sub-subject-en"
   522→                value={newSubmittal.subjectEn}
   523→                onChange={(e) =>
   524→                  setNewSubmittal({ ...newSubmittal, subjectEn: e.target.value })
   525→                }
   526→                placeholder="e.g. Concrete mix design — grade C40/50"
   527→                autoFocus
   528→              />
   529→            </div>
   530→            <div className="space-y-1.5">
   531→              <Label htmlFor="sub-subject-ar">{t("subjectAr")}</Label>
   532→              <Input
   533→                id="sub-subject-ar"
   534→                value={newSubmittal.subjectAr}
   535→                onChange={(e) =>
   536→                  setNewSubmittal({ ...newSubmittal, subjectAr: e.target.value })
   537→                }
   538→                dir="rtl"
   539→              />
   540→            </div>
   541→            <div className="grid grid-cols-2 gap-3">
   542→              <div className="space-y-1.5">
   543→                <Label htmlFor="sub-type">{t("type")}</Label>
   544→                <Select
   545→                  value={newSubmittal.type}
   546→                  onValueChange={(v) =>
   547→                    setNewSubmittal({
   548→                      ...newSubmittal,
   549→                      type: v as SubmittalType,
   550→                    })
   551→                  }
   552→                >
   553→                  <SelectTrigger id="sub-type" className="w-full">
   554→                    <SelectValue />
   555→                  </SelectTrigger>
   556→                  <SelectContent>
   557→                    {SUBMITTAL_TYPES.map((tp) => (
   558→                      <SelectItem key={tp} value={tp}>
   559→                        {tp}
   560→                      </SelectItem>
   561→                    ))}
   562→                  </SelectContent>
   563→                </Select>
   564→              </div>
   565→              <div className="space-y-1.5">
   566→                <Label htmlFor="sub-discipline">{t("discipline")}</Label>
   567→                <Select
   568→                  value={newSubmittal.discipline}
   569→                  onValueChange={(v) =>
   570→                    setNewSubmittal({
   571→                      ...newSubmittal,
   572→                      discipline: v as Discipline,
   573→                    })
   574→                  }
   575→                >
   576→                  <SelectTrigger id="sub-discipline" className="w-full">
   577→                    <SelectValue />
   578→                  </SelectTrigger>
   579→                  <SelectContent>
   580→                    {DISCIPLINES.map((d) => (
   581→                      <SelectItem key={d} value={d}>
   582→                        {d}
   583→                      </SelectItem>

   906→function humanStatus(s: SubmittalStatus): string {
   907→  return s
   908→    .toLowerCase()
   909→    .replace(/_/g, " ")
   910→    .replace(/\b\w/g, (c) => c.toUpperCase());
   911→}

   913→function addDays(iso: string, days: number): string {
   914→  const m = /^(\d{4})-(\d{2})-(\d{2})$/.exec(iso);
   915→  if (!m) return iso;
   916→  const [, y, mo, d] = m;
   917→  const ms = Date.UTC(Number(y), Number(mo) - 1, Number(d));
   918→  return new Date(ms + days * 86400000)
   919→    .toISOString()
   920→    .slice(0, 10);
   921→}
   922→
   923→/**
   924→ * Compute the overdue state per BR-DC6.
   925→ *   overdue ⇔ asOf > due AND status is non-final AND status != DRAFT.
   926→ *   due-soon ⇔ 0 ≤ (due − asOf) ≤ 3 days AND status is non-final AND != DRAFT.
   927→ *
   928→ * The caller passes the injected "today" (BR-DC1 — no Date.now() in domain).
   929→ */
   930→function computeOverdueState(
   931→  asOf: string,
   932→  submittedDate: string | null,
   933→  reviewPeriodDays: number,
   934→  status: SubmittalStatus,
   935→): "overdue" | "due-soon" | "none" {
   936→  if (submittedDate === null) return "none";
   937→  if (status === "DRAFT") return "none";
   938→  if (FINAL_SUBMITTAL_STATUSES.includes(status)) return "none";
   939→
   940→  const due = addDays(submittedDate, reviewPeriodDays);
   941→  const diff = daysBetween(asOf, due);
   942→  if (diff < 0) return "overdue"; // asOf > due
   943→  if (diff <= 3) return "due-soon";
   944→  return "none";
   945→}
   946→
   947→function daysBetween(asOf: string, due: string): number {
   948→  const a = Date.parse(asOf + "T00:00:00Z");
   949→  const b = Date.parse(due + "T00:00:00Z");
   950→  if (Number.isNaN(a) || Number.isNaN(b)) return Number.POSITIVE_INFINITY;
   951→  return Math.round((b - a) / 86400000);
   952→}

   954→function todayIso(): string {
   955→  const d = new Date();
   956→  const y = d.getUTCFullYear();
   957→  const m = String(d.getUTCMonth() + 1).padStart(2, "0");
   958→  const day = String(d.getUTCDate()).padStart(2, "0");
   959→  return `${y}-${m}-${day}`;
   960→}
